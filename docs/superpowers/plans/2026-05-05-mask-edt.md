**Goal:** Replace the binary-fill-then-blur mask generation with a proper Euclidean Distance Transform for accurate, seam-free distance fields.

**Architecture:** Rewrite `generate_mask_data` in `blur.rs` to use Felzenszwalb & Huttenlocher's EDT algorithm. The EDT computes exact squared Euclidean distances to the nearest zero-pixel in two separable passes (rows then columns). A 1-pixel frame of zeros at the mask boundary provides edge conditions for full-window regions. No other files change.

**Tech Stack:** Rust, no new dependencies.

**Spec:** `docs/superpowers/specs/2026-05-05-mask-edt-design.md`

```rust
fn generate_mask_data(
    mask_w: i32,
    mask_h: i32,
    subregion_rects: &[[f32; 4]],
) -> Vec<u8> {
    let w = mask_w as usize;
    let h = mask_h as usize;

    let mut binary = vec![0u8; w * h];

    if subregion_rects.is_empty() {
        for v in &mut binary {
            *v = 1;
        }
    } else {
        for rect in subregion_rects {
            let px1 = (rect[0] * mask_w as f32).floor() as i32;
            let py1 = (rect[1] * mask_h as f32).floor() as i32;
            let px2 = (rect[2] * mask_w as f32).ceil() as i32;
            let py2 = (rect[3] * mask_h as f32).ceil() as i32;

            let px1 = px1.clamp(0, mask_w) as usize;
            let py1 = py1.clamp(0, mask_h) as usize;
            let px2 = px2.clamp(0, mask_w) as usize;
            let py2 = py2.clamp(0, mask_h) as usize;

            for py in py1..py2 {
                let row = py * w;
                for px in px1..px2 {
                    binary[row + px] = 1;
                }
            }
        }
    }

    // Force 1-pixel frame of zeros for boundary condition.
    for x in 0..w {
        binary[x] = 0;
        binary[(h - 1) * w + x] = 0;
    }
    for y in 0..h {
        binary[y * w] = 0;
        binary[y * w + w - 1] = 0;
    }

    // Initialize squared distance: 0 for outside, large value for inside.
    let inf = (w * w + h * h) as f32;
    let mut dist = vec![0.0f32; w * h];
    for i in 0..(w * h) {
        dist[i] = if binary[i] == 1 { inf } else { 0.0 };
    }

    // Row pass: 1D EDT along each row.
    for y in 0..h {
        edt_1d(&mut dist[y * w..(y + 1) * w]);
    }

    // Column pass: 1D EDT along each column.
    let mut col = vec![0.0f32; h];
    for x in 0..w {
        for y in 0..h {
            col[y] = dist[y * w + x];
        }
        edt_1d(&mut col);
        for y in 0..h {
            dist[y * w + x] = col[y];
        }
    }

    // Sqrt to get actual distance, find max for normalization.
    let mut max_dist = 0.0f32;
    for d in &mut dist {
        *d = d.sqrt();
        if *d > max_dist {
            max_dist = *d;
        }
    }

    // Normalize to [0, 1] and convert to bytes.
    let scale = if max_dist > 0.0 { 1.0 / max_dist } else { 0.0 };
    let mut data = vec![0u8; w * h * 4];
    for i in 0..(w * h) {
        let byte = (dist[i] * scale * 255.0).round() as u8;
        let idx = i * 4;
        data[idx] = byte;
        data[idx + 1] = byte;
        data[idx + 2] = byte;
        data[idx + 3] = 255;
    }

    data
}

/// 1D Euclidean Distance Transform using Felzenszwalb & Huttenlocher's parabola
/// envelope method. Transforms squared distances in-place.
fn edt_1d(f: &mut [f32]) {
    let n = f.len();
    if n == 0 {
        return;
    }

    let mut v: Vec<usize> = Vec::with_capacity(n);
    let mut z: Vec<f32> = Vec::with_capacity(n + 1);
    let mut k = 0usize;
    v.push(0);
    z.push(f32::NEG_INFINITY);
    z.push(f32::INFINITY);

    for q in 1..n {
        let fq = f[q];
        while k > 0 {
            let s = ((fq + (q as f32) * (q as f32))
                - (f[v[k]] + (v[k] as f32) * (v[k] as f32)))
                / (2 * q as f32 - 2 * v[k] as f32);
            if s <= z[k] {
                k -= 1;
                v.pop();
                z.pop();
            } else {
                break;
            }
        }
        k += 1;
        v.push(q);
        let s = ((fq + (q as f32) * (q as f32))
            - (f[v[k - 1]] + (v[k - 1] as f32) * (v[k - 1] as f32)))
            / (2 * q as f32 - 2 * v[k - 1] as f32);
        z.push(s);
        z[k + 1] = f32::INFINITY;
    }

    k = 0;
    for q in 0..n {
        while z[k + 1] < q as f32 {
            k += 1;
        }
        let dq = q as f32 - v[k] as f32;
        f[q] = (q as f32 - v[k] as f32) * dq + f[v[k]];
    }
}
```
